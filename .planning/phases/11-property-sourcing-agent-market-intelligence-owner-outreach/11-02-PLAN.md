       export interface OutreachDraftResult {
         contactHistoryId: number;
         channel: 'email' | 'post';
         subject: string;
         content: string;
         pdfBuffer?: Buffer;
       }

       export async function draftOutreach(leadId: number, sequenceStep: number = 0): Promise<OutreachDraftResult>
       - Lazy import pool from server/db, openai from server/lib/openaiClient
       - Query the lead from proactive_leads by id
       - Determine channel from follow-up sequence: step 0 and step 2 = 'post' (letter), step 1 = 'email'
         Default sequence: [{day:0, channel:'post'}, {day:7, channel:'email'}, {day:21, channel:'post'}]
         If lead has campaignId, check lead_monitoring_configs.config.followUpSequence
       - Build source-specific prompt for OpenAI:
         * Context: lead source, property address, postcode, owner name, days on market, original agent, auction result, etc.
         * For expired_listing: "we have active buyers looking in {postcode}... fresh marketing approach"
         * For auction (unsold/withdrawn): "quick private sale... discreet alternative to auction"
         * For land_registry (recent, <2 years): "welcome to the area... complimentary valuation"
         * For land_registry (long-term, >15 years): "properties like yours are in high demand"
         * For land_registry (probate): sensitive approach, "understand this may be a difficult time"
         * For planning_permission: "we noticed planning permission... if you're considering selling"
         * For competitor_listing: "if you're looking for a fresh approach... premium marketing"
         * Tone: professional, local expertise, never pushy, British English
         * Sign off as John Barclay Estate Agents (not as Charlie)
       - Call openai.chat.completions.create with gpt-4o-mini for cost efficiency
       - If channel is 'post': call generateOutreachLetterPDF to create PDF, store in uploads/sourcing/
       - Insert lead_contact_history record:
         * leadId, contactMethod=channel, contactDirection='outbound'
         * subject, content (the AI-drafted text)
         * status='draft', approvalStatus='pending'
         * sequenceStep, pdfUrl (if letter)
       - Update proactive_leads.status to 'ready_to_contact' if currently 'new' or 'researching'
       - Return the draft result

       export async function sendApprovedEmail(contactHistoryId: number): Promise<void>
       - Query lead_contact_history by id, get lead details
       - Call emailService.sendEmail(ownerEmail, subject, content formatted as HTML)
       - Update lead_contact_history: status='sent', sentAt=NOW()
       - Update proactive_leads: status='contacted', contactAttempts++, lastContactDate=NOW(), lastContactMethod='email'
       - Calculate nextFollowUpDate based on sequence step timing

       - Update proactive_leads: status='contacted', contactAttempts++, lastContactDate=NOW(), lastContactMethod='post'
       - Calculate nextFollowUpDate (letter is manually posted by staff who downloaded the PDF)

       export async function getPendingApprovals(): Promise<Array<{...}>>
       - Lazy import pool
       - Query lead_contact_history JOIN proactive_leads WHERE approval_status='pending'
       - Return: contactHistoryId, leadId, channel, subject, content, pdfUrl, sequenceStep,
         lead propertyAddress, postcode, ownerName, leadSource, propensityScore, daysOnMarket
       - ORDER BY created_at ASC (oldest first)

       export async function approveOutreach(contactHistoryId: number, approvedById: number): Promise<void>
       - Lazy import pool, sourcingOutreachService
       - Update lead_contact_history SET approvalStatus='approved', approvedById, approvedAt=NOW()
       - Read the record to determine channel
       - If channel='email': call sendApprovedEmail(contactHistoryId)
       - If channel='post': call sendApprovedLetter(contactHistoryId)
         (PDF already generated; staff downloads from pdfUrl to print and post)
       - Log via auditLogger

       export async function rejectOutreach(contactHistoryId: number, rejectedById: number, reason?: string): Promise<void>
       - Lazy import pool
       - Update lead_contact_history SET approvalStatus='rejected', approvedById=rejectedById, rejectionReason=reason
       - Update proactive_leads.status to 'declined' if no other pending outreach exists for this lead
       - Log via auditLogger

       export async function editOutreachDraft(contactHistoryId: number, newContent: string, newSubject?: string): Promise<void>
       - Lazy import pool
       - Update lead_contact_history SET content=newContent, subject=newSubject
       - If channel='post', regenerate PDF with new content and update pdfUrl

    2. Create server/agents/services/sourcingFollowUpService.ts:

       Default follow-up sequence (used when campaign has no custom sequence):
       const DEFAULT_SEQUENCE = [
         { step: 0, day: 0, channel: 'post' as const },
         { step: 1, day: 7, channel: 'email' as const },
         { step: 2, day: 21, channel: 'post' as const },
       ];

       export function getFollowUpSequence(campaignConfig?: any): FollowUpStep[]
       - If campaignConfig?.followUpSequence exists, return it
       - Otherwise return DEFAULT_SEQUENCE

       export function getNextFollowUpStep(currentStep: number, sequence: FollowUpStep[]): FollowUpStep | null
       - Return sequence[currentStep + 1] or null if exhausted

       export function calculateNextFollowUpDate(currentStep: number, sequence: FollowUpStep[]): Date | null
       - Get next step, if exists: return NOW() + (nextStep.day - currentStep.day) days
       - If no next step: return null (sequence complete)

       export async function advanceFollowUpSequence(leadId: number): Promise<void>
       - Lazy import pool, sourcingOutreachService
       - Query proactive_leads to get current state
       - Get campaign config if campaignId exists
       - Get sequence, determine current step from last contact's sequenceStep
       - Get next step, if exists:
         * Call draftOutreach(leadId, nextStep.step) to create next draft
         * Update proactive_leads.nextFollowUpDate
       - If sequence exhausted: log "sequence complete" to notes
