<!-- API contracts from Plan 03 -->

GET /api/crm/sourcing/leads
Response: {
  stages: {
    new: { count: number, leads: Lead[] },
    researching: { count: number, leads: Lead[] },
    ready_to_contact: { count: number, leads: Lead[] },
    contacted: { count: number, leads: Lead[] },
    responded: { count: number, leads: Lead[] },
    meeting_scheduled: { count: number, leads: Lead[] },
    valuation_booked: { count: number, leads: Lead[] },
    instructed: { count: number, leads: Lead[] },
  },
  total: number
}

Lead: {
  id, leadSource, propertyAddress, postcode, ownerName, ownerEmail,
  leadScore, propensityScore, daysOnMarket, originalAgent,
  status, contactAttempts, lastContactDate, nextFollowUpDate,
  auctionHouse, auctionResult, createdAt
}

GET /api/crm/sourcing/leads/:id
Response: { lead: Lead, contactHistory: ContactHistory[], propensityScore: PropensityScore }

GET /api/crm/sourcing/approvals
Response: Array<{ contactHistoryId, leadId, channel, subject, content, pdfUrl,
  sequenceStep, propertyAddress, postcode, ownerName, leadSource, propensityScore, daysOnMarket }>

POST /api/crm/sourcing/approvals/:id/approve -> { success: true }
POST /api/crm/sourcing/approvals/:id/reject  -> { success: true } (body: { reason? })
PUT  /api/crm/sourcing/approvals/:id/edit    -> { success: true } (body: { content, subject? })

GET /api/crm/sourcing/campaigns -> Array<Campaign>
POST /api/crm/sourcing/campaigns -> Campaign (body: campaign fields)
PUT /api/crm/sourcing/campaigns/:id -> Campaign
DELETE /api/crm/sourcing/campaigns/:id -> { success: true }

GET /api/crm/sourcing/metrics
Response: { leadsSourcedThisMonth, outreachSent, responseRate, valuationsBooked }

GET /api/crm/sourcing/metrics/by-source
Response: Array<{ source, leadsFound, outreachSent, responses, responseRate, valuations, instructions, conversionRate }>
</interfaces>
